// --- 必要な全機能を定義しました --- async function handleSubmit() { const data = { company_name: document.getElementById('company_name').value, closest_station: document.getElementById('closest_station').value, memo: document.getElementById('memo').value, status: document.getElementById('status_select').value }; await fetch(`${API_URL}/api/jobs`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) }); location.reload(); } async function uploadTopFile() { const fileInput = document.getElementById('topFile'); if (!fileInput.files[0]) return alert('ファイルを選択してください'); const formData = new FormData(); formData.append('file', fileInput.files[0]); const res = await fetch(`${API_URL}/api/upload`, { method: 'POST', body: formData }); const data = await res.json(); topUploadedUrl = data.url; alert('アップロード完了'); } function insertTopLink() { const field = document.activeElement; if (!topUploadedUrl) return alert('先にPDFをアップロードしてください'); const start = field.selectionStart; const end = field.selectionEnd; field.value = field.value.substring(0, start) + `[${field.value.substring(start, end)}](${API_URL}${topUploadedUrl})` + field.value.substring(end); } async function handleUpload() { const input = document.getElementById('file-input'); if (!input.files[0]) return; const formData = new FormData(); formData.append('file', input.files[0]); await fetch(`${API_URL}/api/upload`, { method: 'POST', body: formData }); location.reload(); } function handleView() { const name = document.getElementById('common-select').value; if (name) window.open(`${API_URL}/PDF/${encodeURIComponent(name)}`, '_blank'); } async function handleDelete() { const name = document.getElementById('common-select').value; if (name && confirm('削除しますか？')) { await fetch(`${API_URL}/api/files/${encodeURIComponent(name)}`, { method: 'DELETE' }); location.reload(); } }
