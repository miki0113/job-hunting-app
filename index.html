// --- 追加：選択範囲を保持するための変数 --- let lastSelection = { start: 0, end: 0, field: null }; // テキストエリアで文字を選択したらその場所を保存する document.querySelectorAll('textarea').forEach(el => { el.addEventListener('mouseup', () => { lastSelection = { start: el.selectionStart, end: el.selectionEnd, field: el }; }); el.addEventListener('keyup', () => { lastSelection = { start: el.selectionStart, end: el.selectionEnd, field: el }; }); }); // --- リンク化ボタンの修正 --- function insertTopLink() { if (!topUploadedUrl) { alert('先にPDFをアップロードしてください'); return; } // 最後に選択した場所を呼び出す const field = lastSelection.field; const start = lastSelection.start; const end = lastSelection.end; // もしどこも選択されていなければエラー if (!field || start === end) { alert('リンク化したい文字を選択（ドラッグ）してからボタンを押してください！'); return; } // リンク化処理 const text = field.value.substring(start, end); field.value = field.value.substring(0, start) + `[${text}](${API_URL}${topUploadedUrl})` + field.value.substring(end); alert('リンク化しました！'); }
